import json
import time
import boto3
from botocore.exceptions import ClientError

dynamodb = boto3.resource('dynamodb')
table    = dynamodb.Table('PlayerStats')

# Tras este tiempo sin heartbeat de SessionLastSeenAt, otra conexion puede
# tomar el slot. El heartbeat lo refrescan VerifyPlayerStats (al arrancar) y
# SaveGameData (en cada guardado durante la partida), asi que una sesion
# activa nunca queda stale; solo las zombies (crash, cierre forzado, red caida).
STALE_SESSION_SECONDS = 90


def lambda_handler(event, context):
    try:
        user_id = event['requestContext']['authorizer']['claims']['cognito:username']
        body    = json.loads(event['body'])
        action  = body.get('action', 'register')

        if action == 'unregister':
            return handle_unregister(user_id, body)
        return handle_register(user_id, body)
    except Exception as e:
        print(f"RegisterSession error critico: {e}")
        return api_response(500, "ERROR", "Error interno del servidor.")


def handle_register(user_id, body):
    new_session_token = body.get('sessionToken')
    if not new_session_token:
        return api_response(400, "ERROR", "Falta el token.")

    # Defensa en profundidad: corta a las cuentas baneadas antes de
    # concederles el slot, sin esperar al verify de arranque.
    existing = table.get_item(Key={'UserId': user_id}).get('Item')
    if existing and existing.get('PermanentBan', False):
        print(f"REGISTRO BLOQUEADO: {user_id} esta baneado permanentemente.")
        return api_response(200, "PERMANENT_BANNED",
                            "Cuenta baneada permanentemente.")

    now          = int(time.time())
    stale_cutoff = now - STALE_SESSION_SECONDS

    # Check-and-set atomico. Se puede tomar el slot si:
    #  - no hay sesion activa, o el token esta vacio,
    #  - el slot ya es mio (reclamo idempotente: evita auto-bloqueo si el
    #    cliente reintenta el register con el mismo token),
    #  - la sesion esta stale (sin heartbeat reciente -> sesion zombie).
    # NO se acepta "SessionLastSeenAt no existe" como motivo para tomar el
    # slot: cubria un bug del cliente que borraba el campo y permitia hijack.
    try:
        table.update_item(
            Key={'UserId': user_id},
            UpdateExpression=(
                "SET ActiveSessionToken = :t, "
                "SessionStartedAt = :now, "
                "SessionLastSeenAt = :now "
                "REMOVE SessionIntrusionAttemptAt, "
                "SessionIntrusionNoticeId"
            ),
            ConditionExpression=(
                "attribute_not_exists(ActiveSessionToken) "
                "OR ActiveSessionToken = :empty "
                "OR ActiveSessionToken = :t "
                "OR SessionLastSeenAt < :stale"
            ),
            ExpressionAttributeValues={
                ':t':     new_session_token,
                ':empty': "",
                ':now':   now,
                ':stale': stale_cutoff,
            }
        )
    except ClientError as e:
        if e.response['Error']['Code'] == 'ConditionalCheckFailedException':
            print(f"INTRUSO BLOQUEADO: {user_id} intento entrar desde otro PC.")
            notify_active_session(user_id, new_session_token, now, stale_cutoff)
            return api_response(200, "SESSION_IN_USE",
                                "Ya hay una partida en curso en otro dispositivo.")
        print(f"register error: {e}")
        return api_response(500, "ERROR", "Error interno del servidor.")

    return api_response(200, "OK", "Sesion registrada.")


def notify_active_session(user_id, blocked_token, now, stale_cutoff):
    try:
        table.update_item(
            Key={'UserId': user_id},
            UpdateExpression=(
                "SET SessionIntrusionAttemptAt = :now, "
                "SessionIntrusionNoticeId = :notice"
            ),
            ConditionExpression=(
                "attribute_exists(ActiveSessionToken) "
                "AND ActiveSessionToken <> :empty "
                "AND ActiveSessionToken <> :blocked "
                "AND SessionLastSeenAt >= :stale"
            ),
            ExpressionAttributeValues={
                ':now': now,
                ':notice': f"{now}:{blocked_token}",
                ':empty': "",
                ':blocked': blocked_token,
                ':stale': stale_cutoff,
            }
        )
    except ClientError as e:
        if e.response['Error']['Code'] != 'ConditionalCheckFailedException':
            print(f"No se pudo notificar intento de intrusion para {user_id}: {e}")


def handle_unregister(user_id, body):
    client_token = body.get('sessionToken', '')

    # sessionToken obligatorio: sin auth de propiedad no se libera nada.
    # La red de seguridad para crashes/force-quits es el stale-takeover
    # de handle_register, no un unregister sin token.
    if not client_token:
        print(f"WARNING unregister sin sessionToken (user={user_id}).")
        return api_response(200, "IGNORED",
                            "Falta el token de sesion. Se ignora la peticion.")

    now          = int(time.time())
    stale_cutoff = now - STALE_SESSION_SECONDS

    # Se libera el slot si se cumple CUALQUIERA de estas:
    #   1. El cliente posee el token actual (caso normal: el dueno cierra).
    #   2. El slot ya esta vacio / no existe (idempotente, evita 500 raros).
    #   3. El token guardado es zombie (sin heartbeat hace > STALE segs).
    # La (3) es la clave del bug del "doble jugador": cuando un conflicto
    # deja un token huerfano (p.ej. el intruso tomo el slot por stale y luego
    # se fue sin unregister), el siguiente cliente legitimo de la misma
    # cuenta Cognito puede limpiarlo en vez de esperar otros 90 s.
    # Sigue siendo seguro: si el token de DB esta fresco y no es el tuyo,
    # la condicion falla y no se toca nada -> nadie puede tirar la sesion
    # viva de otro PC.
    try:
        table.update_item(
            Key={'UserId': user_id},
            UpdateExpression=(
                "REMOVE ActiveSessionToken, "
                "SessionStartedAt, "
                "SessionLastSeenAt"
            ),
            ConditionExpression=(
                "attribute_not_exists(ActiveSessionToken) "
                "OR ActiveSessionToken = :empty "
                "OR ActiveSessionToken = :t "
                "OR SessionLastSeenAt < :stale"
            ),
            ExpressionAttributeValues={
                ':empty': "",
                ':t':     client_token,
                ':stale': stale_cutoff,
            }
        )
        print(f"unregister OK para {user_id}. Slot liberado.")
        return api_response(200, "OK", "Cuenta liberada correctamente.")

    except ClientError as e:
        # Solo se llega aqui si el slot esta fresco con un token distinto:
        # no es tuyo y no es zombie. No se toca nada.
        if e.response['Error']['Code'] == 'ConditionalCheckFailedException':
            print(f"unregister ALREADY_FREE para {user_id} "
                  f"(slot fresco con token distinto).")
            return api_response(200, "ALREADY_FREE",
                                "El slot pertenece a otra sesion activa.")
        print(f"unregister error: {e}")
        return api_response(500, "ERROR", "Error interno del servidor.")


def api_response(status, code, message):
    return {
        'statusCode': status,
        'headers': {
            'Access-Control-Allow-Origin': '*',
            'Content-Type': 'application/json',
        },
        'body': json.dumps({'code': code, 'message': message}),
    }
